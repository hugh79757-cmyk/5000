### Task 3: single.html 교체
**파일**: 5개 사이트의 `layouts/_default/single.html`

#### 변경 전 (현재)
```go
{{ $parts := split $content "<h2" }}
```

#### 변경 후 (교체)
```go
{{ $content := .Content }}
{{ $adCount := 0 }}

{{/* 조건부 배치 결정 */}}
{{ if ge .WordCount 800 }}
  {{ if in (slice "cars" "finance" "insurance" "crypto" "loans" "health") .Section }}
    {{ $adCount = 3 }}
  {{ else }}
    {{ $adCount = 2 }}
  {{ end }}
{{ end }}

{{/* 마커 주입 */}}
{{ if ge $adCount 1 }}
  {{/* 첫 </p> 직후 */}}
  {{ $content = replace $content "</p>" "</p><!--AD_INARTICLE_1-->" 1 }}
{{ end }}

{{ if ge $adCount 2 }}
  {{/* 첫 번째 <h2> 앞 */}}
  {{ $content = replace $content "<h2" "<!--AD_INARTICLE_2--><h2" 1 }}
{{ end }}

{{ if ge $adCount 3 }}
  {{/* 두 번째 <h2> 앞 (첫 h2에는 이미 마커가 있으므로 카운트) */}}
  {{ $content = replace $content "<h2" "<!--AD_INARTICLE_3--><h2" 2 }}
{{ end }}

{{/* 리더보드 (본문 끝) */}}
{{ $content = printf "%s%s" $content (partial "adsense/leaderboard.html" .) }}

{{/* 마커 → 파셜 치환 */}}
{{ $content = replace $content "<!--AD_INARTICLE_1-->" (partial "adsense/in-article.html" .) | safeHTML }}
{{ $content = replace $content "<!--AD_INARTICLE_2-->" (partial "adsense/in-article.html" .) | safeHTML }}
{{ $content = replace $content "<!--AD_INARTICLE_3-->" (partial "adsense/in-article.html" .) | safeHTML }}

{{ $content | safeHTML }}
```

**확인**: 5개 사이트 모두 동일한 single.html 적용

### Task 4: baseof.html 수정
**파일**: 5개 사이트의 `layouts/_default/baseof.html` (없으면 생성)

`</body>` 직전에 추가:

```html
{{/* === AdSense 모바일 스티키 === */}}
{{ partial "adsense/mobile-sticky.html" . }}

{{/* === AdSense 글로벌 lazy load === */}}
<script>
(function() {
  // 자동광고가 adsbygoogle.js를 이미 로드했는지 확인
  if (window.adsbygoogle) {
    // IntersectionObserver로 lazyad 관찰
    var io = new IntersectionObserver(function(entries) {
      entries.forEach(function(entry) {
        if (entry.isIntersecting) {
          var el = entry.target;
          // 이미 초기화된 ins는 건너뛰기
          if (el.getAttribute('data-adsbygoogle-status')) return;
          try {
            (window.adsbygoogle = window.adsbygoogle || []).push({});
            el.setAttribute('data-adsbygoogle-status', 'done');
          } catch (e) {}
          io.unobserve(el);
        }
      });
    }, { rootMargin: '400px 0px', threshold: 0 });

    document.querySelectorAll('ins.adsbygoogle.lazyad:not([data-adsbygoogle-status])').forEach(function(ins) {
      io.observe(ins);
    });
  }

  // 모바일 스티키: 스크롤 600px 이후 표시
  var stickyAd = document.getElementById('mobile-sticky-ad');
  if (stickyAd) {
    window.addEventListener('scroll', function() {
      if (window.scrollY > 600) {
        stickyAd.style.display = 'block';
        // 스티키 광고도 push
        if (window.adsbygoogle && !stickyAd.querySelector('ins').getAttribute('data-adsbygoogle-status')) {
          try {
            (window.adsbygoogle = window.adsbygoogle || []).push({});
            stickyAd.querySelector('ins').setAttribute('data-adsbygoogle-status', 'done');
          } catch (e) {}
        }
      }
    }, { passive: true });
  }
})();
</script>
```

**확인**: 5개 사이트 모두 동일한 baseof.html 적용
