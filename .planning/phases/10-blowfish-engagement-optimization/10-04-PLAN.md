### Data Flow for Chart

**Current flow:**
1. `pipeline.py:266` → `generate_content(data, blog_id)` returns dict with `body_md`
2. `pipeline.py:303` → `publish(blog_id, title, body_md, ..., sigungu=data.get("sigungu",""))`
3. `publisher.py:953` → `_write_hugo_post(blog_cfg, title, body_md, slug, category, tags, thumbnail_url, is_draft)`

**Problem:** `_write_hugo_post()` receives only `body_md` (string), not the raw item data with camping numeric fields.

**Solution:** Embed chart data as a JSON HTML comment in the pipeline step before publish, then have post-processor extract and replace it with the `{{< chart >}}` shortcode.

### Blowfish Chart Shortcode Format

```yaml
{{< chart >}}
type: 'radar',
data: {
  labels: ['일반야영장', '오토캠핑', '글램핑', '카라반'],
  datasets: [{
    label: '사이트 수',
    data: [20, 10, 5, 8],
    backgroundColor: 'rgba(59, 130, 246, 0.2)',
    borderColor: '#3b82f6',
    borderWidth: 2
  }]
},
options: {
  scales: { r: { beginAtZero: true, ticks: { stepSize: 5 } } },
  plugins: { legend: { display: false } }
}
{{< /chart >}}
```

The chart shortcode uses inline YAML-like format (not strict JSON) — it's parsed internally by Hugo into Chart.js config.

### _clean_body() Compatibility

The `_clean_body()` regex `\{\{(?![<%])[^}]+\}\}`:
- Preserves `{{< chart >}}` (has `<` after `{{`)
- Preserves content inside `{{< chart >}}...{{< /chart >}}`
- Would strip bare `{{ }}` template vars (correct behavior)

    def _apply_chart_shortcode(body_md: str) -> str:
        if not body_md or "{{< chart >}}" in body_md:
            return body_md
        
        import json, re
        
        m = re.search(r"<!-- CHART_DATA:\s*(.+?)-->", body_md, re.DOTALL)
        if not m:
            return body_md
        
        try:
            chart_data = json.loads(m.group(1).strip())
        except (json.JSONDecodeError, Exception):
            return body_md
        
        if not chart_data or not isinstance(chart_data, list):
            return body_md
        
        # Check if any item has numeric data
        has_data = any(
            any(v for k, v in item.items() if k != "name" and isinstance(v, (int, float)))
            for item in chart_data
        )
        if not has_data:
            return re.sub(r"<!-- CHART_DATA:.*?-->", "", body_md)
        
        # Build chart config
        if len(chart_data) == 1:
            chart_config = _build_single_bar_chart(chart_data[0])
        else:
            chart_config = _build_multi_radar_chart(chart_data)
        
        # Replace comment with chart shortcode
        chart_block = f"{{{{< chart >}}}}\n{chart_config}\n{{{{< /chart >}}}}"
        body_md = re.sub(r"<!-- CHART_DATA:.*?-->", chart_block, body_md, flags=re.DOTALL)
        
        return body_md
    
    
    def _build_single_bar_chart(item: dict) -> str:
        FIELD_LABELS = {
            "gnrlSiteCo": "일반야영장",
            "autoSiteCo": "오토캠핑",
            "glampSiteCo": "글램핑",
            "caravSiteCo": "카라반",
            "toiletCo": "화장실",
            "swrmCo": "샤워실",
            "wtrplCo": "개수대",
        }
        labels = []
        data = []
        for field, label in FIELD_LABELS.items():
            val = item.get(field, 0)
            if isinstance(val, (int, float)) and val > 0:
                labels.append(label)
                data.append(val)
        
        if not data:
            return ""
        
        labels_str = "[" + ", ".join(f"'{l}'" for l in labels) + "]"
        data_str = "[" + ", ".join(str(d) for d in data) + "]"
        
        return f"""type: 'bar',
    data: {{
      labels: {labels_str},
      datasets: [{{
        label: '{item.get("name", "캠핑장")}',
        data: {data_str},
        backgroundColor: 'rgba(59, 130, 246, 0.5)',
        borderColor: '#3b82f6',
        borderWidth: 2
      }}]
    }},
    options: {{
      scales: {{ y: {{ beginAtZero: true, ticks: {{ stepSize: 5 }} }} }},
      plugins: {{ legend: {{ display: false }} }}
    }}"""
    
    
    def _build_multi_radar_chart(items: list) -> str:
        FIELD_LABELS = {
            "gnrlSiteCo": "일반야영장",
            "autoSiteCo": "오토캠핑",
            "glampSiteCo": "글램핑",
            "caravSiteCo": "카라반",
            "toiletCo": "화장실",
            "swrmCo": "샤워실",
            "wtrplCo": "개수대",
        }
        COLORS = [
            ("rgba(59, 130, 246, 0.2)", "#3b82f6"),
            ("rgba(239, 68, 68, 0.2)", "#ef4444"),
            ("rgba(34, 197, 94, 0.2)", "#22c55e"),
            ("rgba(234, 179, 8, 0.2)", "#eab308"),
        ]
        
        labels = [label for _, label in FIELD_LABELS.items()]
        labels_str = "[" + ", ".join(f"'{l}'" for l in labels) + "]"
        
        datasets = []
        for i, item in enumerate(items):
            name = item.get("name", f"캠핑장 {i+1}")
            bg, border = COLORS[i % len(COLORS)]
            data = []
            for field in FIELD_LABELS:
                val = item.get(field, 0)
                data.append(val if isinstance(val, (int, float)) else 0)
            data_str = "[" + ", ".join(str(d) for d in data) + "]"
            datasets.append(f"""{{
            label: '{name}',
            data: {data_str},
            backgroundColor: '{bg}',
            borderColor: '{border}',
            borderWidth: 2
          }}""")
        
        datasets_str = "[" + ", ".join(datasets) + "]"
        
        return f"""type: 'radar',
    data: {{
      labels: {labels_str},
      datasets: {datasets_str}
    }},
    options: {{
      scales: {{ r: {{ beginAtZero: true, ticks: {{ stepSize: 5 }} }} }},
      plugins: {{ legend: {{ position: 'bottom' }} }}
    }}"""
